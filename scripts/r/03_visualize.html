ALL_GROUPS <- c("Material", "Pawn structure", "Positional", "Coordination & development")

FEATURE_ORDER <- c(MATERIAL_FEATURES, PAWN_FEATURES, POSITIONAL_FEATURES, DEVELOPMENT_FEATURES)

FEATURE_GROUP <- tibble(
  feature_name = FEATURE_ORDER,
  group = c(
    rep("Material",                   length(MATERIAL_FEATURES)),
    rep("Pawn structure",             length(PAWN_FEATURES)),
    rep("Positional",                 length(POSITIONAL_FEATURES)),
    rep("Coordination & development", length(DEVELOPMENT_FEATURES))
  )
)

# Canonical piece values (for reference lines in material section)
CANONICAL <- tibble(
  feature_name  = MATERIAL_FEATURES,
  canonical_val = c(9, 5, 3, 3, 1)
)

# Outcome → numeric from white's perspective
outcome_numeric <- function(outcome) {
  case_when(
    outcome == "1-0"     ~  1,
    outcome == "0-1"     ~ -1,
    outcome == "1/2-1/2" ~  0,
    .default = NA_real_
  )
}

# Rolling correlation helper
roll_cor <- function(x, y, window = 100) {
  n <- length(x)
  map_dbl(seq_len(n), function(i) {
    if (i < window) return(NA_real_)
    cor(x[(i - window + 1):i], y[(i - window + 1):i], use = "complete.obs")
  })
}

Weight Trajectories

How each feature weight evolves over the course of training. Features are grouped by type. Weights start near zero and should drift toward values that reflect the true importance of each chess concept.

Code 
weights_tidy <- weights |>
  left_join(FEATURE_GROUP, by = "feature_name") |>
  mutate(
    feature_name = factor(feature_name, levels = FEATURE_ORDER),
    group        = factor(group, levels = ALL_GROUPS)
  )

plot_weight_group <- function(grp) {
  weights_tidy |>
    filter(group == grp) |>
    ggplot(aes(x = game_number, y = weight_value, colour = feature_name)) +
    geom_line(linewidth = 0.8) +
    geom_hline(yintercept = 0, linetype = "dashed", colour = "grey60") +
    facet_wrap(~ run_id, labeller = label_both, scales = "free_x") +
    labs(
      title  = paste(grp, "weights over training"),
      x      = "Game number",
      y      = "Weight value",
      colour = NULL
    ) +
    theme_minimal(base_size = 12) +
    theme(legend.position = "bottom")
}

Most Impactful Games

Which individual games drove the largest weight change for a given feature? Because temporal difference learning updates weights after every game, some games — depending on their outcome, length, and the positions reached — produce outsized shifts. Identifying these games lets you examine the actual play that taught the engine a concept.

Set FOCUS_FEATURE and TOP_N to explore any feature.

Code 
FOCUS_FEATURE <- "material_queen"
TOP_N         <- 10

Feature-Outcome Correlation

For each feature, does its average value during a game predict whether white wins? A positive correlation means: higher average feature value (from white’s perspective) is associated with white winning.

This is the primary tool for testing the pawn U-curve hypothesis: if the pawn feature-outcome correlation dips negative during the phase when queen weights are rising, and then recovers, that is evidence of a structural learning sequence rather than noise.

Code 
if (!has_feature_stats || nrow(feature_stats) == 0) {
  cat("No feature_stats data yet. Run a training session with selfplay.ipynb",
      "to populate this table.\n")
} else {
  ROLL_WIN <- 200

  corr_data <- feature_stats |>
    left_join(FEATURE_GROUP, by = "feature_name") |>
    mutate(
      outcome_num  = outcome_numeric(outcome),
      feature_name = factor(feature_name, levels = FEATURE_ORDER),
      group        = factor(group, levels = ALL_GROUPS)
    ) |>
    arrange(run_id, feature_name, game_number) |>
    group_by(run_id, feature_name, group) |>
    mutate(roll_corr = roll_cor(mean_value, outcome_num, window = ROLL_WIN)) |>
    ungroup()

  corr_data |>
    ggplot(aes(x = game_number, y = roll_corr, colour = feature_name)) +
    geom_line(linewidth = 0.7) +
    geom_hline(yintercept = 0, linetype = "dashed", colour = "grey60") +
    facet_grid(group ~ run_id, scales = "free_x") +
    labs(
      title    = paste0("Rolling feature-outcome correlation (", ROLL_WIN, "-game window)"),
      subtitle = "Positive = higher feature value predicts white wins",
      x        = "Game number",
      y        = "Correlation with outcome",
      colour   = NULL
    ) +
    theme_minimal(base_size = 12) +
    theme(legend.position = "bottom")
}

Cross-Run Comparison

Concept emergence timing across independent runs — the central research question. A concept is considered “emerged” when its weight crosses a defined threshold for the first time.

Code 
EMERGENCE_THRESHOLD <- 0.1

emergence <- weights |>
  left_join(FEATURE_GROUP, by = "feature_name") |>
  filter(weight_value > EMERGENCE_THRESHOLD) |>
  group_by(run_id, feature_name, group) |>
  summarise(emerged_at = min(game_number), .groups = "drop") |>
  mutate(
    feature_name = factor(feature_name, levels = FEATURE_ORDER),
    group        = factor(group, levels = ALL_GROUPS)
  )

if (nrow(emergence) == 0) {
  cat("No features have yet crossed the emergence threshold (", EMERGENCE_THRESHOLD, ").\n",
      "Run more training games.\n")
} else {
  emergence |>
    ggplot(aes(x = emerged_at, y = feature_name, colour = factor(run_id))) +
    geom_point(size = 3) +
    facet_wrap(~ group, scales = "free_y", ncol = 1) +
    labs(
      title    = paste0("Concept emergence (weight > ", EMERGENCE_THRESHOLD, ")"),
      subtitle = "Each dot = first checkpoint where weight crossed threshold",
      x        = "Game number",
      y        = NULL,
      colour   = "Run"
    ) +
    theme_minimal(base_size = 12) +
    theme(legend.position = "bottom")
}
